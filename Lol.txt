  function createGrid() {
    for (let y = 0; y < GRID_SIZE; y++) {
      cells[y] = [];
      for (let x = 0; x < GRID_SIZE; x++) {
        const cellEl = document.createElement('div');
        cellEl.className = 'cell';
        cellEl.dataset.x = x;
        cellEl.dataset.y = y;
        gridEl.appendChild(cellEl);

        const cell = {
          x,
          y,
          el: cellEl,
          type: 'empty',
          gateType: null,
          orientation: 0,
          color: null,
          powered: false,
          on: false,
          delay: 4,
          delayCounter: 0,
          pulsePhase: 0,
          pulseOn: 5,
          pulseOff: 5,
          memoryState: false,
          memoryColor: null,
          intensity: 0,
          layerColors: {},
          leverColors: {} // color -> true
        };
        cells[y][x] = cell;

        cellEl.addEventListener('mousedown', (e) => {
          isMouseDown = true;
          mouseButton = e.button;
          handleCellMouseDown(cell, e);
        });
        cellEl.addEventListener('mouseenter', (e) => {
          if (currentMode === 'erase' && isMouseDown && mouseButton === 0) {
            eraseCell(cell);
          }
          if (currentMode === 'select' && isMouseDown && mouseButton === 0 && selectionStart) {
            updateSelectionRect(cell);
          }
        });
        cellEl.addEventListener('mouseup', () => {
          isMouseDown = false;
        });
      }
    }
  }

  function handleCellMouseDown(cell, e) {
    if (currentMode === 'drag') return;

    if (currentMode === 'erase' && e.button === 0) {
      pushHistory('erase');
      eraseCell(cell);
      return;
    }

    if (currentMode === 'clear-all') {
      pushHistory('clear-all');
      clearAll();
      return;
    }

    if (currentMode === 'select') {
      if (e.button === 0) {
        selectionStart = cell;
        selectionRect = { x1: cell.x, y1: cell.y, x2: cell.x, y2: cell.y };
        updateSelectionOverlay();
      } else if (e.button === 2 && selectionCells.length > 0) {
        isDraggingSelection = true;
      }
      return;
    }

    if (currentMode === 'autoroute') {
      if (!autorouteStart) {
        autorouteStart = cell;
      } else {
        pushHistory('autoroute');
        autoRoute(autorouteStart, cell);
        autorouteStart = null;
      }
      return;
    }

    if (currentMode === 'rotate') {
      if (['gate', 'power', 'lever', 'lamp', 'delay', 'pulse', 'memory', 'wire-layer'].includes(cell.type)) {
        cell.orientation = (cell.orientation + 1) % 4;
      }
      return;
    }

    if (currentMode === 'mouse') {
      if (cell.type === 'lever') {
        cell.on = !cell.on;
        return;
      }
      if (cell.type === 'delay') {
        openDelayPopup(cell);
        return;
      }
      if (cell.type === 'pulse') {
        openPulsePopup(cell);
        return;
      }
      if (cell.type === 'memory') {
        return;
      }
      pushHistory('edit');
      placeTool(cell);
    }
  }

      const dy = e.clientY - dragStartY;
      gridContainer.scrollLeft = dragScrollLeft - dx;
      gridContainer.scrollTop = dragScrollTop - dy;

        isDraggingView = false;

  function placeTool(cell) {
    if (currentTool === 'wire-colored') {
      if (cell.type === 'wire') {
        cell.color = activeColor;
        lastColor = activeColor;
        updateLastColorBox();
        return;
      }
      cell.type = 'wire';
      cell.color = activeColor;
      cell.gateType = null;
      cell.on = false;
      return;
    }

    if (currentTool === 'wire-universal') {
      cell.type = 'wire_universal';
      cell.color = null;
      cell.gateType = null;
      cell.on = false;
      return;
    }

    if (currentTool === 'wire-layer') {
      cell.type = 'wire-layer';
      cell.layerColors = {};
      return;
    }

    switch (currentTool) {
      case 'gate-and':
        cell.type = 'gate';
        cell.gateType = 'and';
        cell.on = false;
        break;
      case 'gate-or':
        cell.type = 'gate';
        cell.gateType = 'or';
        cell.on = false;
        break;
      case 'gate-xor':
        cell.type = 'gate';
        cell.gateType = 'xor';
        cell.on = false;
        break;
      case 'delay-block':
        cell.type = 'delay';
        cell.delay = 4;
        cell.delayCounter = 0;
        break;
      case 'pulse-generator':
        cell.type = 'pulse';
        cell.pulseOn = 5;
        cell.pulseOff = 5;
        cell.pulsePhase = 0;
        break;
      case 'memory-cell':
        cell.type = 'memory';
        cell.memoryState = false;
        cell.memoryColor = null;
        break;
      case 'power':
        cell.type = 'power';
        cell.intensity = 20;
        break;
      case 'lever':
        cell.type = 'lever';
        cell.on = false;
        cell.leverColors = {};
        break;
      case 'lamp':
        cell.type = 'lamp';
        cell.on = false;
        break;
      default:
        break;
    }
  }

  function neighbors(x, y) {
    const res = [];
    if (x > 0) res.push({ dir: 'left', cell: cells[y][x - 1] });
    if (x < GRID_SIZE - 1) res.push({ dir: 'right', cell: cells[y][x + 1] });
    if (y > 0) res.push({ dir: 'up', cell: cells[y - 1][x] });
    if (y < GRID_SIZE - 1) res.push({ dir: 'down', cell: cells[y + 1][x] });
    return res;
  }

  function isWire(cell) {
    return cell.type === 'wire' || cell.type === 'wire_universal';
  }

  function wireColorConnects(a, b) {
    if (!isWire(a) || !isWire(b)) return true;
    if (a.type === 'wire_universal' || b.type === 'wire_universal') return true;
    if (!a.color || !b.color) return false;
    return a.color.toLowerCase() === b.color.toLowerCase();
  }

  function isBlockSideInput(block, side) {
    if (block.type === 'gate' || block.type === 'delay' || block.type === 'pulse') {
      const o = block.orientation;
      if (o === 0 || o === 2) {
        return side === 'up' || side === 'down';
      } else if (o === 1 || o === 3) {
        return side === 'left' || side === 'right';
      }
    } else if (block.type === 'lever') {
      const o = block.orientation;
      if (o === 0) return side === 'left';
      if (o === 1) return side === 'up';
      if (o === 2) return side === 'right';
      if (o === 3) return side === 'down';
    } else if (block.type === 'power') {
      return true;
    } else if (block.type === 'lamp') {
      return true;
    } else if (block.type === 'memory') {
      const io = memoryIOSides(block);
      if (side === io.dataSide || io.triggerSides.includes(side)) return true;
    }
    return false;
  }

  function isBlockSideOutput(block, side) {
    if (block.type === 'gate' || block.type === 'delay' || block.type === 'pulse') {
      const o = block.orientation;
      if (o === 0) return side === 'right';
      if (o === 1) return side === 'down';
      if (o === 2) return side === 'left';
      if (o === 3) return side === 'up';
    } else if (block.type === 'lever') {
      const o = block.orientation;
      if (o === 0) return side === 'right';
      if (o === 1) return side === 'down';
      if (o === 2) return side === 'left';
      if (o === 3) return side === 'up';
    } else if (block.type === 'power') {
      return true;
    } else if (block.type === 'lamp') {
      return false;
    } else if (block.type === 'memory') {
      const io = memoryIOSides(block);
      return side === io.outputSide;
    }
    return false;
  }

  function memoryIOSides(mem) {
    const o = mem.orientation;
    if (o === 0) {
      return {
        outputSide: 'up',
        triggerSides: ['left', 'right'],
        dataSide: 'down'
      };
    } else if (o === 1) {
      return {
        outputSide: 'right',
        triggerSides: ['up', 'down'],
        dataSide: 'left'
      };
    } else if (o === 2) {
      return {
        outputSide: 'down',
        triggerSides: ['left', 'right'],
        dataSide: 'up'
      };
    } else {
      return {
        outputSide: 'left',
        triggerSides: ['up', 'down'],
        dataSide: 'right'
      };
    }
  }

  function oppositeSide(side) {
    if (side === 'left') return 'right';
    if (side === 'right') return 'left';
    if (side === 'up') return 'down';
    if (side === 'down') return 'up';
    return null;
  }

  function simulate() {
    if (simulationFrozen) {
      render();
      return;
    }

    for (let y = 0; y < GRID_SIZE; y++) {
      for (let x = 0; x < GRID_SIZE; x++) {
        const c = cells[y][x];
        c.powered = false;
        c.intensity = 0;
        if (c.type === 'lamp') c.on = false;
      }
    }

    for (let y = 0; y < GRID_SIZE; y++) {
      for (let x = 0; x < GRID_SIZE; x++) {
        const c = cells[y][x];
        if (c.type === 'power') {
          c.powered = true;
          c.intensity = 20;
        } else if (c.type === 'pulse') {
          const period = c.pulseOn + c.pulseOff;
          if (period <= 0) continue;
          c.pulsePhase = (c.pulsePhase + 1) % period;
          c.powered = c.pulsePhase < c.pulseOn;
          c.intensity = c.powered ? 15 : 0;
        } else if (c.type === 'memory') {
          if (c.memoryState) {
            c.powered = true;
            c.intensity = 10;
          }
        }
      }
    }

    propagateWiresWithIntensity();

    const gateOutputs = [];
    const leverOutputs = [];

    for (let y = 0; y < GRID_SIZE; y++) {
      for (let x = 0; x < GRID_SIZE; x++) {
        const c = cells[y][x];

        if (c.type === 'gate') {
          const inputsPowered = getGateInputsPowered(c);
          const out = evalGate(c.gateType, inputsPowered.count);
          if (out) {
            const outNeighbor = getGateOutputNeighbor(c);
            if (outNeighbor && isWire(outNeighbor.cell) && wireColorConnects(outNeighbor.cell, c)) {
              gateOutputs.push({ cell: outNeighbor.cell, intensity: inputsPowered.maxIntensity });
            }
          }
        } else if (c.type === 'delay') {
          const inputsPowered = getGateInputsPowered(c);
          if (inputsPowered.count > 0) {
            if (!c.delayCounter || c.delayCounter <= 0) c.delayCounter = c.delay;
            c.delayCounter--;
            if (c.delayCounter <= 0) {
              const outNeighbor = getGateOutputNeighbor(c);
              if (outNeighbor && isWire(outNeighbor.cell) && wireColorConnects(outNeighbor.cell, c)) {
                gateOutputs.push({ cell: outNeighbor.cell, intensity: inputsPowered.maxIntensity });
              }
              c.delayCounter = c.delay;
            }
          } else {
            c.delayCounter = 0;
          }
        } else if (c.type === 'memory') {
          const memInputs = getMemoryInputs(c);
          if (memInputs.trigger) {
            c.memoryState = memInputs.dataPowered;
            c.memoryColor = memInputs.dataColor;
          }
        } else if (c.type === 'lever') {
          updateLeverColorsFromInput(c);
          if (c.on) {
            const outNeighbor = getLeverOutputNeighbor(c);
            if (outNeighbor && isWire(outNeighbor.cell)) {
              const colors = Object.keys(c.leverColors);
              colors.forEach(col => {
                const inputPowered = isLeverColorPowered(c, col);
                if (inputPowered) {
                  leverOutputs.push({ cell: outNeighbor.cell, color: col, intensity: 10 });
                }
              });
            }
          }
        }
      }
    }

    gateOutputs.forEach(o => {
      o.cell.powered = true;
      o.cell.intensity = Math.max(o.cell.intensity, o.intensity);
    });

    leverOutputs.forEach(o => {
      if (o.cell.type === 'wire') {
        if (o.cell.color && o.cell.color.toLowerCase() === o.color.toLowerCase()) {
          o.cell.powered = true;
          o.cell.intensity = Math.max(o.cell.intensity, o.intensity);
        }
      } else if (o.cell.type === 'wire_universal') {
        o.cell.powered = true;
        o.cell.intensity = Math.max(o.cell.intensity, o.intensity);
      }
    });

    propagateWiresWithIntensity();

    for (let y = 0; y < GRID_SIZE; y++) {
      for (let x = 0; x < GRID_SIZE; x++) {
        const c = cells[y][x];
        if (c.type === 'lamp') {
          const neigh = neighbors(c.x, c.y);
          let on = false;
          for (const n of neigh) {
            const nb = n.cell;
            if (isWire(nb) && nb.powered) {
              on = true;
              break;
            }
            if ((nb.type === 'power' || nb.type === 'lever' || nb.type === 'gate' || nb.type === 'delay' || nb.type === 'pulse' || nb.type === 'memory') &&
                nb.powered &&
                isBlockSideOutput(nb, oppositeSide(n.dir))) {
              on = true;
              break;
            }
          }
          c.on = on;
        }
      }
    }

    propagateLampAdjacency();
    render();
  }

  function propagateWiresWithIntensity() {
    let changed = true;
    while (changed) {
      changed = false;
      for (let y = 0; y < GRID_SIZE; y++) {
        for (let x = 0; x < GRID_SIZE; x++) {
          const c = cells[y][x];
          if (!isWire(c)) continue;
          const neigh = neighbors(x, y);
          let bestIntensity = c.intensity;
          let shouldPower = c.powered;
          for (const n of neigh) {
            const nb = n.cell;
            if (isWire(nb) && wireColorConnects(c, nb) && nb.powered) {
              if (nb.intensity > bestIntensity) {
                bestIntensity = nb.intensity;
                shouldPower = true;
              }
            }
            if ((nb.type === 'power' || nb.type === 'lever' || nb.type === 'gate' || nb.type === 'delay' || nb.type === 'pulse' || nb.type === 'memory') &&
                nb.powered &&
                isBlockSideOutput(nb, oppositeSide(n.dir))) {
              if (nb.intensity > bestIntensity) {
                bestIntensity = nb.intensity;
                shouldPower = true;
              }
            }
            if (nb.type === 'wire-layer') {
              const color = c.color;
              if (!color) continue;
              nb.layerColors[color.toLowerCase()] = true;
              const otherNeighbors = neighbors(nb.x, nb.y);
              otherNeighbors.forEach(onb => {
                const oc = onb.cell;
                if (oc === c) return;
                if (!isWire(oc)) return;
                if (!oc.color) return;
                if (oc.color.toLowerCase() !== color.toLowerCase()) return;
                if (oc.powered && oc.intensity > bestIntensity) {
                  bestIntensity = oc.intensity;
                  shouldPower = true;
                }
              });
            }
          }
          if (shouldPower && bestIntensity > c.intensity) {
            c.powered = true;
            c.intensity = bestIntensity;
            changed = true;
          }
        }
      }
    }
  }

  function getGateInputsPowered(gate) {
    const x = gate.x;
    const y = gate.y;
    const neigh = neighbors(x, y);
    let count = 0;
    let maxIntensity = 0;
    neigh.forEach(n => {
      const nb = n.cell;
      const side = n.dir;
      if (!isWire(nb)) return;
      if (!wireColorConnects(nb, gate)) return;
      if (!isBlockSideInput(gate, side)) return;
      if (nb.powered) {
        count++;
        if (nb.intensity > maxIntensity) maxIntensity = nb.intensity;
      }
    });
    return { count, maxIntensity };
  }

  function getGateOutputNeighbor(gate) {
    const x = gate.x;
    const y = gate.y;
    const neigh = neighbors(x, y);
    for (const n of neigh) {
      const side = n.dir;
      if (isBlockSideOutput(gate, side)) {
        return n;
      }
    }
    return null;
  }

  function getMemoryInputs(mem) {
    const x = mem.x;
    const y = mem.y;
    const neigh = neighbors(x, y);
    const io = memoryIOSides(mem);
    let trigger = false;
    let dataPowered = false;
    let dataColor = null;

    neigh.forEach(n => {
      const nb = n.cell;
      const side = n.dir;
      if (!isWire(nb)) return;
      if (io.triggerSides.includes(side)) {
        if (nb.powered) trigger = true;
      }
      if (side === io.dataSide) {
        if (nb.powered) {
          dataPowered = true;
          if (nb.color) dataColor = nb.color;
        } else {
          dataPowered = false;
        }
      }
    });

    return { trigger, dataPowered, dataColor };
  }

  function evalGate(type, inputsPoweredCount) {
    const n = inputsPoweredCount;
    if (type === 'and') return n >= 2;
    if (type === 'or') return n >= 1;
    if (type === 'xor') return n === 1;
    return false;
  }

  function propagateLampAdjacency() {
    const visited = [];
    for (let y = 0; y < GRID_SIZE; y++) {
      visited[y] = [];
      for (let x = 0; x < GRID_SIZE; x++) {
        visited[y][x] = false;
      }
    }

    for (let y = 0; y < GRID_SIZE; y++) {
      for (let x = 0; x < GRID_SIZE; x++) {
        const c = cells[y][x];
        if (c.type !== 'lamp' || visited[y][x]) continue;

        const stack = [c];
        const group = [];
        let anyOn = false;
        visited[y][x] = true;

        while (stack.length > 0) {
          const cur = stack.pop();
          group.push(cur);
          if (cur.on) anyOn = true;
          const neigh = neighbors(cur.x, cur.y);
          for (const n of neigh) {
            const nb = n.cell;
            if (nb.type === 'lamp' && !visited[nb.y][nb.x]) {
              visited[nb.y][nb.x] = true;
              stack.push(nb);
            }
          }
        }

        if (anyOn) {
          group.forEach(l => { l.on = true; });
        }
      }
    }
  }

  function updateLeverColorsFromInput(lever) {
    const neigh = neighbors(lever.x, lever.y);
    const o = lever.orientation;
    let inputSide = null;
    if (o === 0) inputSide = 'left';
    else if (o === 1) inputSide = 'up';
    else if (o === 2) inputSide = 'right';
    else if (o === 3) inputSide = 'down';

    lever.leverColors = {};
    neigh.forEach(n => {
      const nb = n.cell;
      const side = n.dir;
      if (side !== inputSide) return;
      if (!isWire(nb)) return;
      if (nb.color) {
        lever.leverColors[nb.color.toLowerCase()] = true;
      }
    });
  }

  function isLeverColorPowered(lever, color) {
    const neigh = neighbors(lever.x, lever.y);
    const o = lever.orientation;
    let inputSide = null;
    if (o === 0) inputSide = 'left';
    else if (o === 1) inputSide = 'up';
    else if (o === 2) inputSide = 'right';
    else if (o === 3) inputSide = 'down';

    let powered = false;
    neigh.forEach(n => {
      const nb = n.cell;
      const side = n.dir;
      if (side !== inputSide) return;
      if (!isWire(nb)) return;
      if (!nb.color) return;
      if (nb.color.toLowerCase() !== color.toLowerCase()) return;
      if (nb.powered) powered = true;
    });
    return powered;
  }

  function getLeverOutputNeighbor(lever) {
    const neigh = neighbors(lever.x, lever.y);
    const o = lever.orientation;
    let outputSide = null;
    if (o === 0) outputSide = 'right';
    else if (o === 1) outputSide = 'down';
    else if (o === 2) outputSide = 'left';
    else if (o === 3) outputSide = 'up';

    for (const n of neigh) {
      if (n.dir === outputSide) return n;
    }
    return null;
  }

  function autoRoute(start, end) {
    const queue = [start];
    const prev = {};
    const key = (x, y) => x + ',' + y;
    prev[key(start.x, start.y)] = null;

    while (queue.length) {
      const c = queue.shift();
      if (c === end) break;
      neighbors(c.x, c.y).forEach(n => {
        const nb = n.cell;
        const k = key(nb.x, nb.y);
        if (!prev.hasOwnProperty(k) && nb.type === 'empty') {
          prev[k] = c;
          queue.push(nb);
        }
      });
    }

    let cur = end;
    if (!prev.hasOwnProperty(key(end.x, end.y))) return;
    while (cur && cur !== start) {
      cur.type = 'wire';
      cur.color = activeColor;
      cur = prev[key(cur.x, cur.y)];
    }
  }

  function clearAll() {
    for (let y = 0; y < GRID_SIZE; y++) {
      for (let x = 0; x < GRID_SIZE; x++) {
        eraseCell(cells[y][x]);
      }
    }
  }

  function pushHistory(label) {
    const snapshot = {
      label,
      data: cells.map(row => row.map(c => structuredCloneCell(c)))
    };
    undoStack.push(snapshot);
    if (undoStack.length > MAX_HISTORY) undoStack.shift();
    redoStack = [];
    updateTimeline();
  }

  function restoreSnapshot(snapshot) {
    for (let y = 0; y < GRID_SIZE; y++) {
      for (let x = 0; x < GRID_SIZE; x++) {
        const src = snapshot.data[y][x];
        const dst = cells[y][x];
        Object.assign(dst, src, { el: dst.el, x, y });
      }
    }
  }

  function undo() {
    if (undoStack.length === 0) return;
    const snapshot = undoStack.pop();
    redoStack.push(snapshot);
    const prev = undoStack[undoStack.length - 1];
    if (prev) restoreSnapshot(prev);
    else clearAll();
    updateTimeline();
  }

  function redo() {
    if (redoStack.length === 0) return;
    const snapshot = redoStack.pop();
    undoStack.push(snapshot);
    restoreSnapshot(snapshot);
    updateTimeline();
  }

  function updateTimeline() {
    timelineEl.innerHTML = '';
    undoStack.forEach((snap, idx) => {
      const item = document.createElement('div');
      item.className = 'timeline-item';
      item.textContent = `${idx}: ${snap.label}`;
      item.addEventListener('click', () => {
        restoreSnapshot(snap);
      });
      timelineEl.appendChild(item);
    });
  }

  function saveToTxt() {
    const data = cells.map(row => row.map(c => structuredCloneCell(c)));
    const json = JSON.stringify(data);
    const blob = new Blob([json], { type: 'text/plain' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = 'logic_sandbox_save.txt';
    a.click();
    URL.revokeObjectURL(url);
  }

  function loadFromTxt() {
    const input = document.createElement('input');
    input.type = 'file';
    input.accept = '.txt';
    input.addEventListener('change', () => {
      const file = input.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = () => {
        try {
          const data = JSON.parse(reader.result);
          for (let y = 0; y < GRID_SIZE; y++) {
            for (let x = 0; x < GRID_SIZE; x++) {
              const src = data[y][x];
              const dst = cells[y][x];
              Object.assign(dst, src, { el: dst.el, x, y });
            }
          }
          pushHistory('load');
        } catch (e) {
          console.error('Invalid save file', e);
        }
      };
      reader.readAsText(file);
    });
    input.click();
  }

  function exportPNG() {
    const cellSize = parseInt(getComputedStyle(document.documentElement).getPropertyValue('--cell-size'), 10);
    const canvas = document.createElement('canvas');
    canvas.width = GRID_SIZE * cellSize;
    canvas.height = GRID_SIZE * cellSize;
    const ctx = canvas.getContext('2d');

    ctx.fillStyle = '#101010';
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    for (let y = 0; y < GRID_SIZE; y++) {
      for (let x = 0; x < GRID_SIZE; x++) {
        const c = cells[y][x];
        const px = x * cellSize;
        const py = y * cellSize;
        if (c.type === 'empty') continue;
        if (c.type === 'wire') {
          ctx.fillStyle = c.color || '#ffffff';
          ctx.fillRect(px + cellSize/4, py + cellSize/4, cellSize/2, cellSize/2);
        } else if (c.type === 'wire_universal') {
          ctx.fillStyle = '#aaaaaa';
          ctx.fillRect(px + cellSize/4, py + cellSize/4, cellSize/2, cellSize/2);
        } else if (c.type === 'wire-layer') {
          ctx.fillStyle = '#555';
          ctx.fillRect(px + 4, py + 4, cellSize - 8, cellSize - 8);
        } else {
          ctx.fillStyle = '#444';
          ctx.fillRect(px+1, py+1, cellSize-2, cellSize-2);
        }
      }
    }

    const url = canvas.toDataURL('image/png');
    const a = document.createElement('a');
    a.href = url;
    a.download = 'logic_sandbox.png';
    a.click();
  }

  function updateSelectionOverlay() {
    if (!selectionRect) {
      selectionOverlay.style.display = 'none';
      return;
    }
    const x1 = Math.min(selectionRect.x1, selectionRect.x2);
    const x2 = Math.max(selectionRect.x1, selectionRect.x2);
    const y1 = Math.min(selectionRect.y1, selectionRect.y2);
    const y2 = Math.max(selectionRect.y1, selectionRect.y2);

    const cellSize = parseInt(getComputedStyle(document.documentElement).getPropertyValue('--cell-size'), 10);
    const left = x1 * cellSize + gridEl.getBoundingClientRect().left - gridContainer.getBoundingClientRect().left;
    const top = y1 * cellSize + gridEl.getBoundingClientRect().top - gridContainer.getBoundingClientRect().top;
    const width = (x2 - x1 + 1) * cellSize;
    const height = (y2 - y1 + 1) * cellSize;

    selectionOverlay.style.left = left + 'px';
    selectionOverlay.style.top = top + 'px';
    selectionOverlay.style.width = width + 'px';
    selectionOverlay.style.height = height + 'px';
    selectionOverlay.style.display = 'block';
  }

  function updateSelectionRect(cell) {
    selectionRect.x2 = cell.x;
    selectionRect.y2 = cell.y;
    updateSelectionOverlay();
  }

  function finalizeSelection() {
    if (!selectionRect) return;
    const x1 = Math.min(selectionRect.x1, selectionRect.x2);
    const x2 = Math.max(selectionRect.x1, selectionRect.x2);
    const y1 = Math.min(selectionRect.y1, selectionRect.y2);
    const y2 = Math.max(selectionRect.y1, selectionRect.y2);
    selectionCells = [];
    for (let y = y1; y <= y2; y++) {
      for (let x = x1; x <= x2; x++) {
        selectionCells.push(cells[y][x]);
      }
    }
  }

  window.addEventListener('keydown', (e) => {
    if (e.ctrlKey && e.key.toLowerCase() === 'z') {
      e.preventDefault();
      undo();
    } else if (e.ctrlKey && e.key.toLowerCase() === 'y') {
      e.preventDefault();
      redo();
    } else if (e.key.toLowerCase() === 'delete' && currentMode === 'select') {
      if (selectionCells.length > 0) {
        pushHistory('delete-selection');
        selectionCells.forEach(c => eraseCell(c));
      }
    }
  });

  gridContainer.addEventListener('mousemove', (e) => {
    if (currentMode === 'select' && isDraggingSelection && selectionRect) {
      const cellSize = parseInt(getComputedStyle(document.documentElement).getPropertyValue('--cell-size'), 10);
      const dx = Math.round(e.movementX / cellSize);
      const dy = Math.round(e.movementY / cellSize);
      selectionRect.x1 += dx;
      selectionRect.x2 += dx;
      selectionRect.y1 += dy;
      selectionRect.y2 += dy;
      updateSelectionOverlay();
    }
  });
